local PlaceId = 109983668079237
local WEBHOOK_URL = "https://discord.com/api/webhooks/1452980356624941158/5zWtp5RnudhTIisPPfjVBAw1iE80ikWT6vLK8LtzP8nCRkVQYYrL6solJ2cqv73bWFT-"
local task = task
local lastJobId = "---"
local sentJobIds = {}
local autoSend = false
local autoRunning = false
local function sendWebhook(jobId)
    if sentJobIds[jobId] then return end
    sentJobIds[jobId] = true
    lastJobId = jobId
    jobLabel.Text = "Last JobId: " .. jobId
    status.Text = "Status: Sending JobId..."
    pcall(function()
        -- Check if request function exists
        if typeof(request) == "function" then
            request({
                Url = WEBHOOK_URL,
                Method = "POST",
                Headers = {["Content-Type"] = "application/json"},
                Body = HttpService:JSONEncode({
                    content = "@here",
                    embeds = {{
                        title = "Kurd Hub Server Found",
                        description = "## Job Id:\n```" .. jobId .. "```",
                        color = 0xAA82FF
                    }}
                })
            })
        end
    end)
    status.Text = "Status: Sent JobId " .. jobId
end
local function findRandomServer()
    local cursor = ""
    local servers = {}
    repeat
        local url = "https://games.roblox.com/v1/games/" .. PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"
        if cursor ~= "" then 
            url = url .. "&cursor=" .. cursor 
        end
        local ok, res = pcall(function()
            local response = game:HttpGet(url)
            return HttpService:JSONDecode(response)
        end)
        if ok and res and res.data then
            for _, s in pairs(res.data) do
                if s.playing < s.maxPlayers and not sentJobIds[s.id] then
                    table.insert(servers, s.id)
                end
            end
            cursor = res.nextPageCursor or ""
        else 
            break 
        end
    until cursor == "" or #servers >= 10
    if #servers > 0 then
        local jobId = servers[math.random(1, #servers)]
        sendWebhook(jobId)
    else
        status.Text = "Status: No servers found"
    end
end
spawn(function()
    while true do
        task.wait(1)
        if autoSend and not autoRunning then
            autoRunning = true
            status.Text = "Status: Searching servers..."
            findRandomServer()
            task.wait(5)
            autoRunning = false
        end
    end
end)
